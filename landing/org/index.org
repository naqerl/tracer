with near-zero overhead.  It streams data back to the panel over a
secure, lightweight protocol.

<script>
(function() {
  function resize(iframe) {
    var doc;
    try { doc = iframe.contentDocument || iframe.contentWindow.document; } catch(e) {}
    if (!doc) return;
    iframe.style.height = doc.documentElement.scrollHeight + 'px';
  }

  document.querySelectorAll('iframe').forEach(function(iframe) {
    function handleLoad() {
      resize(iframe);

      var doc;
      try { doc = iframe.contentDocument || iframe.contentWindow.document; } catch(e) {}
      if (!doc) return;

      new MutationObserver(function() {
        clearTimeout(iframe._rt);
        iframe._rt = setTimeout(function() { resize(iframe); }, 50);
      }).observe(doc.body, { childList: true, subtree: true });
    }

    iframe.addEventListener('load', handleLoad);

    var doc;
    try { doc = iframe.contentDocument || iframe.contentWindow.document; } catch(e) {}
    if (doc && doc.readyState === 'complete') handleLoad();
  });
})();
</script>
